-- Google BigQuery Partitioned Telemetry Table (asia-south1)
CREATE TABLE IF NOT EXISTS `kavachgrid-prod.coldchain_telemetry.fleet_events`
(
  event_id STRING NOT NULL,
  unit_id STRING NOT NULL,
  corridor_id STRING NOT NULL,
  recorded_at TIMESTAMP NOT NULL,
  chamber_temp_c FLOAT64,
  ambient_temp_c FLOAT64,
  cryo_battery_pct INT64,
  sha256_packet_hash STRING,
  autonomous_action_dispatched STRING
)
PARTITION BY DATE(recorded_at)
CLUSTER BY corridor_id, unit_id;

-- Vertex AI Feature View Query: Detect Thermal Drift > 0.08 C/min
SELECT
  unit_id,
  corridor_id,
  AVG(chamber_temp_c) OVER (PARTITION BY unit_id ORDER BY recorded_at ROWS BETWEEN 5 PRECEDING AND CURRENT ROW) AS rolling_temp_c,
  ML.DETECT_ANOMALIES(MODEL `kavachgrid-prod.vertex_models.thermal_drift_arima`, STRUCT(0.95 AS contamination))
FROM `kavachgrid-prod.coldchain_telemetry.fleet_events`
WHERE recorded_at >= TIMESTAMP_SUB(CURRENT_TIMESTAMP(), INTERVAL 30 MINUTE);
